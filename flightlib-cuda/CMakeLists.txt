cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" FLIGHTMARE_VERSION LIMIT_COUNT 1)
project(flightlib_cuda VERSION ${FLIGHTMARE_VERSION} LANGUAGES CXX)

add_library(flightlib_cuda_contracts INTERFACE)
add_library(flightlib_cuda::contracts ALIAS flightlib_cuda_contracts)
target_include_directories(flightlib_cuda_contracts INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
set_target_properties(flightlib_cuda_contracts PROPERTIES EXPORT_NAME contracts)
target_compile_features(flightlib_cuda_contracts INTERFACE cxx_std_17)

include(CTest)
set(FLIGHTLIB_CUDA_ORACLE_LIBRARY "" CACHE FILEPATH "Previously built frozen CPU oracle library")
option(FLIGHTLIB_CUDA_TEST_DEVICE "Compile and exercise contracts with CUDA" OFF)
option(FLIGHTLIB_CUDA_BUILD_RUNTIME "Build the CUDA storage and launch runtime" OFF)
if(FLIGHTLIB_CUDA_BUILD_RUNTIME OR FLIGHTLIB_CUDA_TEST_DEVICE)
  enable_language(CUDA)
  set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
  find_package(CUDAToolkit 13.0 REQUIRED)
  find_package(Eigen3 REQUIRED)
  find_package(yaml-cpp REQUIRED)
  add_library(flightlib_cuda_runtime STATIC src/runtime.cu src/model.cpp src/lifecycle.cu src/dynamics.cu src/operations.cu src/environment.cu)
  set_source_files_properties(src/operations.cu PROPERTIES COMPILE_OPTIONS "--fmad=false")
  set_source_files_properties(src/dynamics.cu PROPERTIES COMPILE_OPTIONS "--fmad=false;-Xptxas=-v")
  set_source_files_properties(src/environment.cu PROPERTIES COMPILE_OPTIONS "--fmad=false;-Xptxas=-v;--expt-relaxed-constexpr")
  add_library(flightlib_cuda::runtime ALIAS flightlib_cuda_runtime)
  target_link_libraries(flightlib_cuda_runtime PUBLIC flightlib_cuda::contracts CUDA::cudart
    PRIVATE CUDA::cuda_driver Eigen3::Eigen yaml-cpp::yaml-cpp)
  set_target_properties(flightlib_cuda_runtime PROPERTIES
    CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON EXPORT_NAME runtime)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS flightlib_cuda_contracts EXPORT flightlib_cudaTargets)
if(TARGET flightlib_cuda_runtime)
  install(TARGETS flightlib_cuda_runtime EXPORT flightlib_cudaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/flightlib_cuda DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT flightlib_cudaTargets NAMESPACE flightlib_cuda:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/flightlib_cuda)
configure_package_config_file(cmake/flightlib_cudaConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/flightlib_cudaConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/flightlib_cuda)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/flightlib_cudaConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/flightlib_cudaConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/flightlib_cudaConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/flightlib_cuda)
option(FLIGHTLIB_CUDA_BUILD_PYTHON "Build Torch interoperability module (no libtorch link)" OFF)
if(FLIGHTLIB_CUDA_BUILD_PYTHON)
  if(NOT TARGET flightlib_cuda_runtime)
    message(FATAL_ERROR "Python bindings require FLIGHTLIB_CUDA_BUILD_RUNTIME")
  endif()
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python COMPONENTS Interpreter Development REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  execute_process(COMMAND ${Python_EXECUTABLE} -c "import torch,pathlib; print(pathlib.Path(torch.__file__).parent/'include')"
    OUTPUT_VARIABLE TORCH_HEADERS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  pybind11_add_module(_native python/bindings.cpp)
  target_include_directories(_native PRIVATE "${TORCH_HEADERS}")
  target_link_libraries(_native PRIVATE flightlib_cuda::runtime)
  set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/flightlib_cuda")
  file(GLOB course_assets CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/python/flightlib_cuda/assets"
       "python/flightlib_cuda/assets/*")
  foreach(asset IN LISTS course_assets)
    configure_file("python/flightlib_cuda/assets/${asset}" "python/flightlib_cuda/assets/${asset}" COPYONLY)
  endforeach()
  configure_file(python/flightlib_cuda/scene.py python/flightlib_cuda/scene.py COPYONLY)
  install(DIRECTORY python/flightlib_cuda/assets DESTINATION flightlib_cuda)
  install(FILES python/flightlib_cuda/scene.py DESTINATION flightlib_cuda)
  configure_file(python/flightlib_cuda/__init__.py python/flightlib_cuda/__init__.py COPYONLY)
  install(TARGETS _native LIBRARY DESTINATION flightlib_cuda)
  install(FILES python/flightlib_cuda/__init__.py DESTINATION flightlib_cuda)
endif()

  target_link_libraries(contracts_test PRIVATE flightlib_cuda::contracts)

      target_compile_definitions(runtime_test PRIVATE FLC_TRACE_RUNTIME_CALLS=1)
      foreach(api cudaMalloc cudaFree cudaMallocAsync cudaFreeAsync cudaMemcpy cudaMemcpyAsync
                  cudaDeviceSynchronize cudaStreamSynchronize cudaEventSynchronize)
        target_link_options(runtime_test PRIVATE "LINKER:--wrap=${api}")
      endforeach()

    set_tests_properties(flightlib_cuda.runtime PROPERTIES TIMEOUT 60)

    target_link_libraries(guarded_test PRIVATE flightlib_cuda::runtime)
    add_test(NAME flightlib_cuda.guarded COMMAND guarded_test)
    set_tests_properties(flightlib_cuda.guarded PROPERTIES TIMEOUT 120)

      target_compile_definitions(dynamics_test PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
      target_compile_options(dynamics_test PRIVATE -march=native -ffp-contract=off)
      target_link_libraries(dynamics_test PRIVATE flightlib_cuda::runtime
        "${FLIGHTLIB_CUDA_ORACLE_LIBRARY}" Eigen3::Eigen yaml-cpp::yaml-cpp
        OpenMP::OpenMP_CXX ${OpenCV_LIBRARIES} zmq zmqpp stdc++fs)
      add_test(NAME flightlib_cuda.dynamics_parity COMMAND dynamics_test)
      set_tests_properties(flightlib_cuda.dynamics_parity PROPERTIES TIMEOUT 60
        ENVIRONMENT "FLIGHTMARE_PATH=${CMAKE_CURRENT_SOURCE_DIR}/..")
      add_executable(environment_test tests/environment_test.cpp)
      target_include_directories(environment_test PRIVATE ../flightlib/include ${OpenCV_INCLUDE_DIRS})
      target_compile_definitions(environment_test PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
      target_compile_options(environment_test PRIVATE -march=native -ffp-contract=off)
      target_link_libraries(environment_test PRIVATE flightlib_cuda::runtime
        "${FLIGHTLIB_CUDA_ORACLE_LIBRARY}" Eigen3::Eigen yaml-cpp::yaml-cpp
        OpenMP::OpenMP_CXX ${OpenCV_LIBRARIES} zmq zmqpp stdc++fs)
      add_test(NAME flightlib_cuda.environment_parity COMMAND environment_test)

        ENVIRONMENT "FLIGHTMARE_PATH=${CMAKE_CURRENT_SOURCE_DIR}/..")
    endif()
  endif()
