# Standalone benchmark target. Link the existing optimized CPU environment for its dependencies.
cmake_minimum_required(VERSION 3.24)
project(course_profile LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Benchmark build type" FORCE)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(root "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(inc "${CMAKE_CURRENT_BINARY_DIR}/course_instrumented.inc")
add_custom_command(OUTPUT "${inc}" COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/instrument.py" "${root}" "${inc}"
  DEPENDS instrument.py "${root}/flightlib/src/objects/quadrotor.cpp" "${root}/flightlib/src/envs/quadrotor_env/quadrotor_env.cpp")
add_library(course_profile SHARED cpu_course.cpp "${inc}")
set(PROFILE_BUILD /build/flightmare-profile CACHE PATH "Existing optimized baseline build")
target_compile_features(course_profile PRIVATE cxx_std_17)
target_include_directories(course_profile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${root}/flightlib/include"
  "${root}/flightlib-cuda/include" "${root}/flightlib-cuda/src" ${OpenCV_INCLUDE_DIRS} ${CUDAToolkit_INCLUDE_DIRS})
target_compile_options(course_profile PRIVATE -O3 -fno-fast-math -ffp-contract=off -march=native)
target_compile_definitions(course_profile PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
target_link_libraries(course_profile PRIVATE "${PROFILE_BUILD}/libcpu_environment.so" Eigen3::Eigen OpenMP::OpenMP_CXX yaml-cpp::yaml-cpp)
target_link_options(course_profile PRIVATE -Wl,-Bsymbolic-functions)
