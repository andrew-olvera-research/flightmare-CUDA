cmake_minimum_required(VERSION 3.24)
project(flightrl_cuda LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(rl_kernels SHARED native/kernels.cu)
target_link_libraries(rl_kernels PRIVATE CUDA::cudart)
set_target_properties(rl_kernels PROPERTIES CUDA_STANDARD 17 POSITION_INDEPENDENT_CODE ON PREFIX "" OUTPUT_NAME "_kernels")
target_compile_options(rl_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ptxas-options=-v>)
if(RL_EXTENSION_NAME)
  set_target_properties(rl_kernels PROPERTIES OUTPUT_NAME "${RL_EXTENSION_NAME}" SUFFIX "")
endif()
