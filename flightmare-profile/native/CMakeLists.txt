cmake_minimum_required(VERSION 3.24)
project(flightmare_profile LANGUAGES CXX CUDA)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FLIGHTLIB_CUDA_BUILD_RUNTIME ON CACHE BOOL "" FORCE)
set(FLIGHTLIB_CUDA_TEST_DEVICE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../flightlib-cuda runtime)
add_executable(profile_driver driver.cu)
target_link_libraries(profile_driver PRIVATE flightlib_cuda::runtime)
set_target_properties(profile_driver PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
foreach(api cudaMalloc cudaFree cudaMallocAsync cudaFreeAsync cudaMemcpy cudaMemcpyAsync
            cudaDeviceSynchronize cudaStreamSynchronize cudaEventSynchronize cudaLaunchKernel __cudaLaunchKernel
            cudaGraphLaunch cudaStreamWaitEvent cudaEventRecord cudaGetDevice cudaSetDevice cudaStreamGetDevice
            cudaStreamGetFlags cudaStreamIsCapturing cudaGetLastError)
  target_link_options(profile_driver PRIVATE "LINKER:--wrap=${api}")
endforeach()

option(PROFILE_CPU_BASELINE "Build frozen sequential CPU baseline" OFF)
if(PROFILE_CPU_BASELINE)
  find_package(Eigen3 REQUIRED)
  find_package(yaml-cpp REQUIRED)
  find_package(OpenCV REQUIRED)
  find_package(OpenMP REQUIRED)
  file(GLOB CPU_SOURCES ../../flightlib/src/common/*.cpp ../../flightlib/src/dynamics/*.cpp ../../flightlib/src/sensors/*.cpp)
  add_library(cpu_optimized STATIC ${CPU_SOURCES} ../../flightlib/src/objects/quadrotor.cpp ../../flightlib/src/objects/object_base.cpp)
  target_include_directories(cpu_optimized PRIVATE ../../flightlib/include ${OpenCV_INCLUDE_DIRS})
  target_compile_definitions(cpu_optimized PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
  target_compile_options(cpu_optimized PRIVATE -O3 -fno-fast-math -ffp-contract=off -march=native)
  target_link_libraries(cpu_optimized PRIVATE Eigen3::Eigen yaml-cpp::yaml-cpp OpenMP::OpenMP_CXX ${OpenCV_LIBRARIES})
  add_executable(cpu_baseline ../../flightlib-cuda/tests/dynamics_test.cpp)
  target_include_directories(cpu_baseline PRIVATE ../../flightlib/include ${OpenCV_INCLUDE_DIRS})
  target_compile_definitions(cpu_baseline PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
  target_compile_options(cpu_baseline PRIVATE -march=native)
  target_link_libraries(cpu_baseline PRIVATE flightlib_cuda::runtime cpu_optimized
    Eigen3::Eigen yaml-cpp::yaml-cpp OpenMP::OpenMP_CXX ${OpenCV_LIBRARIES} zmq zmqpp stdc++fs)
  set_target_properties(cpu_optimized PROPERTIES POSITION_INDEPENDENT_CODE ON)
  file(GLOB BENCH_BRIDGES ../../flightlib/src/bridges/*.cpp)
  add_library(cpu_environment SHARED cpu_environment.cpp
    ../../flightlib/src/envs/env_base.cpp ../../flightlib/src/envs/quadrotor_env/quadrotor_env.cpp ${BENCH_BRIDGES})
  target_include_directories(cpu_environment PRIVATE ../../flightlib/include ${OpenCV_INCLUDE_DIRS})
  target_compile_definitions(cpu_environment PRIVATE EIGEN_STACK_ALLOCATION_LIMIT=1048576)
  target_compile_options(cpu_environment PRIVATE -O3 -fno-fast-math -ffp-contract=off -march=native)
  target_link_libraries(cpu_environment PRIVATE cpu_optimized Eigen3::Eigen yaml-cpp::yaml-cpp OpenMP::OpenMP_CXX ${OpenCV_LIBRARIES} zmq zmqpp stdc++fs)
endif()
